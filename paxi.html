Mainnet · Node Setup
Paxi
A Cosmos SDK blockchain providing decentralized infrastructure and staking services.
paxi-mainnet v1.0.6 Gas: 0.05upaxi Port prefix: 40
Step 1
Install Dependencies & Go
# Cập nhật hệ thống
sudo apt update && sudo apt upgrade -y
# Cài dependencies
sudo apt-get install git curl build-essential make jq gcc snapd chrony lz4 tmux unzip make bc -y
Install Go 1.24.6 (amd64)
# Xóa Go cũ
sudo rm -rf $HOME/go && sudo rm -rf /usr/local/go
cd $HOME
# Tải và giải nén Go 1.24.6
curl https://dl.google.com/go/go1.24.6.linux-amd64.tar.gz | sudo tar -C /usr/local -zxvf -
# Thêm Go vào PATH
cat <<'EOF' >> $HOME/.bashrc
export GOROOT=/usr/local/go
export GOPATH=$HOME/go
export PATH=$PATH:/usr/local/go/bin:$HOME/go/bin
export GOPROXY=https://proxy.golang.org.direct
EOF
# Áp dụng biến môi trường
source $HOME/.bashrc
# Kiểm tra phiên bản Go
go version
Step 2
Set Variables & Install Binary
# Đặt prefix cổng
echo "export PAXI_PORT="40"" >> $HOME/.bash_profile
# Áp dụng biến
source $HOME/.bash_profile
Build from source (v1.0.6)
cd $HOME
# Clone source code
git clone https://github.com/paxi-web3/paxi
cd paxi
# Chuyển sang phiên bản v1.0.6
git checkout v1.0.6
# Biên dịch và cài binary
make install
# Xác nhận phiên bản mới
paxid version
Step 3
Configure Node
Initialize & client config
# Khởi tạo node
paxid init Stilucky-node --chain-id paxi-mainnet
# Ghi chain-id vào config
sed -i "s/chain-id = .*/chain-id = \"paxi-mainnet\"/" $(dirname $(which paxid))/paxi/config/client.toml
# Đặt keyring backend
sed -i "s/keyring-backend = .*/keyring-backend = \"os\"/" $(dirname $(which paxid))/paxi/config/client.toml
# Trỏ RPC về cổng tùy chỉnh
sed -i "s/node = .*/node = \"tcp:\/\/localhost:${PAXI_PORT}657\"/" $(dirname $(which paxid))/paxi/config/client.toml
Custom ports
# Đổi cổng REST/gRPC (app.toml)
sed -i.bak -e "s%:1317%:${PAXI_PORT}317%g;s%:8080%:${PAXI_PORT}080%g;s%:9090%:${PAXI_PORT}090%g;s%:9091%:${PAXI_PORT}091%g;s%:8545%:${PAXI_PORT}545%g;s%:8546%:${PAXI_PORT}546%g;s%:6065%:${PAXI_PORT}065%g" $(dirname $(which paxid))/paxi/config/app.toml
# Đổi cổng P2P/RPC (config.toml)
sed -i.bak -e "s%:26658%:${PAXI_PORT}658%g;s%:26657%:${PAXI_PORT}657%g;s%:6060%:${PAXI_PORT}060%g;s%:26656%:${PAXI_PORT}656%g;s%^external_address = \"\"%external_address = \"$(wget -qO- eth0.me):${PAXI_PORT}656\"%;s%:26660%:${PAXI_PORT}660%g" $(dirname $(which paxid))/paxi/config/config.toml
Gas, indexer, pruning & DB
# Gas tối thiểu
sed -i "s/minimum-gas-prices = .*/minimum-gas-prices = \"0.05upaxi\"/" $(dirname $(which paxid))/paxi/config/app.toml
# Tắt indexer
sed -i -e "s/^indexer *=.*/indexer = \"null\"/" $(dirname $(which paxid))/paxi/config/config.toml
# Bật pruning tùy chỉnh
sed -i -e "s/^pruning *=.*/pruning = \"custom\"/" $(dirname $(which paxid))/paxi/config/app.toml
# Giữ 100 block gần nhất
sed -i -e "s/^pruning-keep-recent *=.*/pruning-keep-recent = \"100\"/" $(dirname $(which paxid))/paxi/config/app.toml
# Pruning mỗi 50 block
sed -i -e "s/^pruning-interval *=.*/pruning-interval = \"50\"/" $(dirname $(which paxid))/paxi/config/app.toml
# Đặt DB backend goleveldb
sed -i "s|db_backend =.*|db_backend=\"goleveldb\"|g" "$(dirname $(which paxid))/paxi/config/config.toml"
sed -i "s|app-db-backend =.*|app-db-backend=\"goleveldb\"|g" "$(dirname $(which paxid))/paxi/config/app.toml"
Genesis & address book
# Tải genesis.json
curl -Ls https://file2.node39.top/Mainnet/Paxi/genesis.json > $(dirname $(which paxid))/paxi/config/genesis.json
# Tải addrbook.json
curl -Ls https://file2.node39.top/Mainnet/Paxi/addrbook.json > $(dirname $(which paxid))/paxi/config/addrbook.json

Step 4
Create Service & Start
# Tạo systemd service
sudo tee /etc/systemd/system/paxid.service > /dev/null <<EOF
[Unit]
Description=Paxi Node
After=network-online.target

[Service]
User=$USER
WorkingDirectory=$HOME
ExecStart=$(which paxid) start
Restart=always
RestartSec=3
LimitNOFILE=65535

[Install]
WantedBy=multi-user.target
EOF
# Reload daemon
sudo systemctl daemon-reload
# Bật auto-start
sudo systemctl enable paxid
# Khởi động node
sudo systemctl start paxid
Check sync status
# catching_up=false → đã sync xong
paxid status 2>&1 | jq .SyncInfo

Stop & build new version

git pull
# Chuyển phiên bản mới
git checkout NEW_VERSION
# Biên dịch và cài binary
make install
# Xác nhận phiên bản mới
paxid version
Restart & verify
# Khởi động lại
sudo systemctl restart paxid
# Xem log realtime

Validator
# Tạo validator
paxid tx staking create-validator validator.json --from <wallet> --chain-id paxi-mainnet --gas auto --gas-adjustment 1.5 --fees 5000upaxi -y
# Delegate token
paxid tx staking delegate <valoper> <amount>upaxi --from <wallet> --chain-id paxi-mainnet --fees 5000upaxi -y
# Rút phần thưởng
paxid tx distribution withdraw-rewards <valoper> --from <wallet> --chain-id paxi-mainnet --fees 5000upaxi -y
# Gỡ jailed validator
paxid tx slashing unjail --from <wallet> --chain-id paxi-mainnet --fees 5000upaxi -y
# Bỏ phiếu governance
paxid tx gov vote <id> yes --from <wallet> --chain-id paxi-mainnet --fees 5000upaxi -y
