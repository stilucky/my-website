Lava Network
A modular blockchain powering decentralized access to RPC and data services across chains.
lava-mainnet-1 v5.5.1 Gas: 0.000000001ulava Port prefix: 39
Step 1
Install Dependencies & Go
# Cập nhật hệ thống
sudo apt update && sudo apt upgrade -y
# Cài dependencies
sudo apt-get install git curl build-essential make jq gcc snapd chrony lz4 tmux unzip make bc -y
Install Go 1.24.6 (amd64)
# Xóa Go cũ
sudo rm -rf $HOME/go && sudo rm -rf /usr/local/go
cd $HOME
# Tải và giải nén Go 1.24.6
curl https://dl.google.com/go/go1.24.6.linux-amd64.tar.gz | sudo tar -C /usr/local -zxvf -
# Thêm Go vào PATH
cat <<'EOF' >> $HOME/.bashrc
export GOROOT=/usr/local/go
export GOPATH=$HOME/go
export PATH=$PATH:/usr/local/go/bin:$HOME/go/bin
export GOPROXY=https://proxy.golang.org.direct
EOF
# Áp dụng biến môi trường
source $HOME/.bashrc
# Kiểm tra phiên bản Go
go version
Step 2
Set Variables & Install Binary
# Đặt prefix cổng
echo "export LAVA_PORT="39"" >> $HOME/.bash_profile
# Áp dụng biến
source $HOME/.bash_profile
Build from source (v5.5.1)
export LAVA_BINARY=lavad
cd $HOME
rm -rf lava
# Clone source code
git clone https://github.com/lavanet/lava.git
cd lava
# Chuyển sang phiên bản v5.5.1
git checkout v5.5.1
# Biên dịch và cài binary
make install
# Xác nhận phiên bản mới
lavad version
Step 3
Configure Node
Initialize & client config
# Khởi tạo node
lavad init Stilucky-node --chain-id lava-mainnet-1
# Ghi chain-id vào config
sed -i "s/chain-id = .*/chain-id = \"lava-mainnet-1\"/" $HOME/.lava/config/client.toml
# Đặt keyring backend
sed -i "s/keyring-backend = .*/keyring-backend = \"os\"/" $HOME/.lava/config/client.toml
# Trỏ RPC về cổng tùy chỉnh
sed -i "s/node = .*/node = \"tcp:\/\/localhost:${LAVA_PORT}657\"/" $HOME/.lava/config/client.toml
Custom ports
# Đổi cổng REST/gRPC (app.toml)
sed -i.bak -e "s%:1317%:${LAVA_PORT}317%g;s%:8080%:${LAVA_PORT}080%g;s%:9090%:${LAVA_PORT}090%g;s%:9091%:${LAVA_PORT}091%g;s%:8545%:${LAVA_PORT}545%g;s%:8546%:${LAVA_PORT}546%g;s%:6065%:${LAVA_PORT}065%g" $HOME/.lava/config/app.toml
# Đổi cổng P2P/RPC (config.toml)
sed -i.bak -e "s%:26658%:${LAVA_PORT}658%g;s%:26657%:${LAVA_PORT}657%g;s%:6060%:${LAVA_PORT}060%g;s%:26656%:${LAVA_PORT}656%g;s%^external_address = \"\"%external_address = \"$(wget -qO- eth0.me):${LAVA_PORT}656\"%;s%:26660%:${LAVA_PORT}660%g" $HOME/.lava/config/config.toml
Gas, indexer & pruning
# Gas tối thiểu
sed -i "s/minimum-gas-prices = .*/minimum-gas-prices = \"0.000000001ulava\"/" $HOME/.lava/config/app.toml
# Tắt indexer
sed -i -e "s/^indexer *=.*/indexer = \"null\"/" $HOME/.lava/config/config.toml
# Bật pruning tùy chỉnh
sed -i -e "s/^pruning *=.*/pruning = \"custom\"/" $HOME/.lava/config/app.toml
# Giữ 100 block gần nhất
sed -i -e "s/^pruning-keep-recent *=.*/pruning-keep-recent = \"100\"/" $HOME/.lava/config/app.toml
# Pruning mỗi 50 block
sed -i -e "s/^pruning-interval *=.*/pruning-interval = \"50\"/" $HOME/.lava/config/app.toml
Genesis & address book
# Tải genesis.json
curl -Ls https://file3.node39.top/Mainnet/Lava/genesis.json > $HOME/.lava/config/genesis.json
# Tải addrbook.json
curl -Ls https://file3.node39.top/Mainnet/Lava/addrbook.json > $HOME/.lava/config/addrbook.json
Step 4
Create Service & Start
# Tạo systemd service
sudo tee /etc/systemd/system/lavad.service > /dev/null <<EOF
[Unit]
Description=Lava Chain Validator
After=network-online.target

[Service]
User=$USER
WorkingDirectory=$HOME
ExecStart=$(which lavad) start
Restart=always
RestartSec=3
LimitNOFILE=65535

[Install]
WantedBy=multi-user.target
EOF
# Reload daemon
sudo systemctl daemon-reload
# Bật auto-start
sudo systemctl enable lavad
# Khởi động node
sudo systemctl start lavad
Check sync status
# catching_up=false → đã sync xong
lavad status 2>&1 | jq .SyncInfo

Validator
# Tạo validator
lavad tx staking create-validator validator.json --from <wallet> --chain-id lava-mainnet-1 --gas auto --gas-adjustment 1.5 --fees 1000ulava -y
# Delegate token
lavad tx staking delegate <valoper> <amount>ulava --from <wallet> --chain-id lava-mainnet-1 --fees 1000ulava -y
# Rút phần thưởng
lavad tx distribution withdraw-rewards <valoper> --from <wallet> --chain-id lava-mainnet-1 --fees 1000ulava -y
# Gỡ jailed validator
lavad tx slashing unjail --from <wallet> --chain-id lava-mainnet-1 --fees 1000ulava -y
# Bỏ phiếu governance
lavad tx gov vote <id> yes --from <wallet> --chain-id lava-mainnet-1 --fees 1000ulava -y
